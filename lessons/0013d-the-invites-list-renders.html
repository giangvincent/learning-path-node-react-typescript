The invites list renders

The payoff of the whole series: the pending-invites list renders at /family/:slug/invites, for a slug the client was given rather than one it guessed. One new screen, one link, one route — and the browser walkthrough that proves the whole client is honest.

Mission: ship KinLink Requires: Lesson 13c Reference: 11 Practice: Lesson 13e
The win

Click a family and its pending invitations render — the address resolved by lesson 12's label rule, a shareable invite labelled "Anyone with the link". The client is now honest end to end: no dead /api/people fetch, no invented slug, no second copy of "am I logged in".

Build it

  1. The list screen

    The page that renders invites needs a slug. It gets one from the URL, and the URL is only reachable through a link the family list built from GET /api/families — so the slug is always one the server handed out.

    client/src/InvitesPage.tsx (new) 
    import { useEffect, useState } from "react";
import { useParams } from "react-router-dom";
import type { InvitationResource } from "./contracts";
import { getInvitations } from "./lib/api";

/**
 * The label rule from lesson 12, in one place: a null email is a shareable link,
 * not a person whose identity is being withheld.
 */
function addressee(invitation: InvitationResource): string {
  return invitation.email ?? "Anyone with the link";
}

export default function InvitesPage() {
  const { slug } = useParams<{ slug: string }>();
  const [invitations, setInvitations] = useState<InvitationResource[] | null>(
    null,
  );
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    if (!slug) return;
    getInvitations(slug)
      .then(setInvitations)
      .catch((cause: unknown) =>
        setError(cause instanceof Error ? cause.message : "Unknown error"),
      );
  }, [slug]);

  if (error) return <p role="alert">Failed to load: {error}</p>;
  if (!invitations) return <p>Loading…</p>;
  if (invitations.length === 0) return <p>No pending invitations.</p>;

  return (
    <main style={{ fontFamily: "system-ui, sans-serif", padding: "2rem" }}>
      <h1>Pending invitations</h1>
      <ul>
        {invitations.map((invitation) => (
          <li key={invitation.id}>
            {addressee(invitation)} — {invitation.role} — sent{" "}
            {new Date(invitation.createdAt).toLocaleDateString()}
          </li>
        ))}
      </ul>
    </main>
  );
}

  2. Link to it, and add the route

    Three small edits to App.tsx: import Link and InvitesPage, wrap the family name in a <Link>, and register the route. Here is the finished file — the only lines that differ from lesson 13c are marked:

    client/src/App.tsx (final) 
    import { useEffect, useState } from "react";
import { Link, Route, Routes } from "react-router-dom";      // + Link
import AcceptPage from "./AcceptPage";
import InvitesPage from "./InvitesPage";                      // + new import
import LoginScreen from "./LoginScreen";
import type { FamilyResource } from "./contracts";
import { getFamilies } from "./lib/api";
import { useSession } from "./lib/useSession";

function FamilyList() {
  const [families, setFamilies] = useState<FamilyResource[] | null>(null);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    getFamilies()
      .then(setFamilies)
      .catch((cause: unknown) =>
        setError(cause instanceof Error ? cause.message : "Unknown error"),
      );
  }, []);

  if (error) return <p role="alert">Failed to load: {error}</p>;
  if (!families) return <p>Loading…</p>;

  return (
    <main style={{ fontFamily: "system-ui, sans-serif", padding: "2rem" }}>
      <h1>KinLink</h1>
      <ul>
        {families.map((family) => (
          <li key={family.slug}>
            <Link to={`/family/${family.slug}/invites`}>{family.name}</Link> —
            you are {family.role}
          </li>
        ))}
      </ul>
    </main>
  );
}

function Home() {
  const { state, refresh } = useSession();

  if (state.status === "loading") return <p>Checking your session…</p>;
  if (state.status === "error") return <p role="alert">{state.message}</p>;
  if (state.status === "anonymous") return <LoginScreen onSignedIn={refresh} />;
  return <FamilyList />;
}

export default function App() {
  return (
    <Routes>
      <Route path="/invite/:token" element={<AcceptPage />} />
      <Route path="/family/:slug/invites" element={<InvitesPage />} />  // + route
      <Route path="/" element={<Home />} />
    </Routes>
  );
}
